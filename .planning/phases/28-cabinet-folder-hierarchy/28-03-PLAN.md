<!-- Backend API contracts from Plan 02 -->

Folder API endpoints:
```
GET    /api/v1/folders/                        -> EnvelopeResponse<FolderResponse[]>
POST   /api/v1/folders/                        -> EnvelopeResponse<FolderResponse>  (body: {name, description?})
GET    /api/v1/folders/tree                    -> EnvelopeResponse<FolderTreeNode[]>
GET    /api/v1/folders/{id}                    -> EnvelopeResponse<FolderResponse>
POST   /api/v1/folders/{id}/children           -> EnvelopeResponse<FolderResponse>  (body: {name, description?})
PUT    /api/v1/folders/{id}                    -> EnvelopeResponse<FolderResponse>  (body: {name?, parent_id?})
DELETE /api/v1/folders/{id}                    -> EnvelopeResponse
POST   /api/v1/folders/{id}/copy               -> EnvelopeResponse<FolderResponse>  (body: {destination_parent_id?})
GET    /api/v1/folders/{id}/documents          -> EnvelopeResponse<Document[]> + PaginationMeta
POST   /api/v1/folders/{id}/documents          -> EnvelopeResponse  (body: {document_id})
DELETE /api/v1/folders/{id}/documents/{doc_id} -> EnvelopeResponse
```

Response types:
```typescript
FolderTreeNode: { id: string, name: string, is_cabinet: boolean, document_count: number, children: FolderTreeNode[] }
FolderResponse: { id: string, name: string, description: string|null, parent_id: string|null, is_cabinet: boolean, document_count: number, path: {id: string, name: string}[], created_at: string, updated_at: string, created_by: string|null }
DocumentResponse now includes: folder_ids: string[]
```

   ```tsx
   // State for folder picker
   const [folderPickerOpen, setFolderPickerOpen] = useState(false);
   const queryClient = useQueryClient();

   // Folder names resolution: fetch tree to map IDs to names
   const { data: folderTree } = useQuery({
     queryKey: folderKeys.tree(),
     queryFn: fetchFolderTree,
     enabled: !!document?.folder_ids?.length,
   });

   // Helper to find folder name from tree
   function findFolderName(tree: FolderTreeNode[], id: string): string | null {
     for (const node of tree) {
       if (node.id === id) return node.name;
       const found = findFolderName(node.children, id);
       if (found) return found;
     }
     return null;
   }

   const fileMutation = useMutation({
     mutationFn: (folderId: string) => fileDocument(folderId, document!.id),
     onSuccess: () => {
       queryClient.invalidateQueries({ queryKey: ["documents", documentId] });
       queryClient.invalidateQueries({ queryKey: folderKeys.tree() });
       setFolderPickerOpen(false);
     },
   });

   const unfileMutation = useMutation({
     mutationFn: (folderId: string) => unfileDocument(folderId, document!.id),
     onSuccess: () => {
       queryClient.invalidateQueries({ queryKey: ["documents", documentId] });
       queryClient.invalidateQueries({ queryKey: folderKeys.tree() });
     },
   });
   ```
