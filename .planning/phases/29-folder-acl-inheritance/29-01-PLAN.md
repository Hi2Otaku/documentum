async def create_folder_acl_entry(
    db: AsyncSession,
    folder_id: uuid.UUID,
    principal_id: uuid.UUID,
    principal_type: str,
    permission_level: PermissionLevel,
    user_id: str | None = None,
) -> FolderACL:
    """Create a folder ACL entry, or return existing if duplicate."""
    from app.models.acl import FolderACL
    result = await db.execute(
        select(FolderACL).where(
            FolderACL.folder_id == folder_id,
            FolderACL.principal_id == principal_id,
            FolderACL.principal_type == principal_type,
            FolderACL.permission_level == permission_level.value,
            FolderACL.is_deleted == False,
        )
    )
    existing = result.scalar_one_or_none()
    if existing is not None:
        return existing

    entry = FolderACL(
        folder_id=folder_id,
        principal_id=principal_id,
        principal_type=principal_type,
        permission_level=permission_level.value,
        created_by=user_id,
    )
    db.add(entry)
    await db.flush()
    await db.refresh(entry)

    await create_audit_record(
        db, entity_type="folder_acl", entity_id=str(folder_id),
        action="acl_granted", user_id=user_id,
        after_state={"principal_id": str(principal_id), "principal_type": principal_type, "permission_level": permission_level.value},
    )
    return entry


async def get_folder_acls(db: AsyncSession, folder_id: uuid.UUID) -> list:
    """Return all non-deleted ACL entries for a folder."""
    from app.models.acl import FolderACL
    result = await db.execute(
        select(FolderACL).where(
            FolderACL.folder_id == folder_id,
            FolderACL.is_deleted == False,
        )
    )
    return list(result.scalars().all())


async def remove_folder_acl_entry(db: AsyncSession, folder_id: uuid.UUID, acl_id: uuid.UUID, user_id: str | None = None) -> bool:
    """Remove a specific folder ACL entry by id. Returns True if found and deleted."""
    from app.models.acl import FolderACL
    result = await db.execute(
        select(FolderACL).where(
            FolderACL.id == acl_id,
            FolderACL.folder_id == folder_id,
            FolderACL.is_deleted == False,
        )
    )
    entry = result.scalar_one_or_none()
    if entry is None:
        return False
    await db.delete(entry)
    await create_audit_record(
        db, entity_type="folder_acl", entity_id=str(folder_id),
        action="acl_revoked", user_id=user_id,
        before_state={"principal_id": str(entry.principal_id), "principal_type": entry.principal_type, "permission_level": entry.permission_level},
    )
    return True


async def check_folder_permission(
    db: AsyncSession,
    folder_id: uuid.UUID,
    user_id: uuid.UUID,
    required_level: PermissionLevel,
    is_superuser: bool = False,
) -> bool:
    """Check if user has sufficient permission on a folder via FolderACL (for ACL management gating)."""
    if is_superuser:
        return True
    from app.models.acl import FolderACL
    from app.models.user import user_groups

    # Check direct user entries
    result = await db.execute(
        select(FolderACL).where(
            FolderACL.folder_id == folder_id,
            FolderACL.principal_id == user_id,
            FolderACL.principal_type == "user",
            FolderACL.is_deleted == False,
        )
    )
    for entry in result.scalars().all():
        if has_sufficient_permission(PermissionLevel(entry.permission_level), required_level):
            return True

    # Check group entries
    group_result = await db.execute(select(user_groups.c.group_id).where(user_groups.c.user_id == user_id))
    group_ids = [row[0] for row in group_result.fetchall()]
    if group_ids:
        group_acl_result = await db.execute(
            select(FolderACL).where(
                FolderACL.folder_id == folder_id,
                FolderACL.principal_type == "group",
                FolderACL.principal_id.in_(group_ids),
                FolderACL.is_deleted == False,
            )
        )
        for entry in group_acl_result.scalars().all():
            if has_sufficient_permission(PermissionLevel(entry.permission_level), required_level):
                return True

    return False

if total_entries == 0:
    # No direct ACL entries — check folder ACL inheritance
    from app.models.folder import document_folders, Folder
    from app.models.acl import FolderACL

    # Get all folders this document is filed in
    folder_result = await db.execute(
        select(document_folders.c.folder_id).where(
            document_folders.c.document_id == document_id
        )
    )
    doc_folder_ids = [row[0] for row in folder_result.all()]

    if not doc_folder_ids:
        return True  # Not filed in any folder, no direct ACL -> open access

    # Recursive CTE: walk from document's folders up to all ancestors
    folder_table = Folder.__table__
    anchor = (
        select(folder_table.c.id)
        .where(
            folder_table.c.id.in_(doc_folder_ids),
            folder_table.c.is_deleted == False,
        )
        .cte(name="folder_ancestors", recursive=True)
    )
    anc_alias = anchor.alias("anc")
    recursive_term = select(folder_table.c.id).join(
        anc_alias, folder_table.c.parent_id == anc_alias.c.id
    )
    # NOTE: The recursive term walks UP: we want parents, so join condition is
    # folder_table.c.id == anc_alias.c.parent_id (wrong above)
    # Actually: the recursive CTE walks by matching children's parent_id to the already-found IDs.
    # But we want to walk UP (find parents). So:
    # anchor = the document's folder IDs
    # recursive = find rows where folder_table.c.id == (parent_id of already-found rows)
    # This means we need to track parent_id in the CTE.

    # Corrected CTE approach (matching _get_folder_path pattern):
    anchor = (
        select(
            folder_table.c.id,
            folder_table.c.parent_id,
        )
        .where(
            folder_table.c.id.in_(doc_folder_ids),
            folder_table.c.is_deleted == False,
        )
        .cte(name="folder_ancestors", recursive=True)
    )
    anc_alias = anchor.alias("anc")
    recursive_term = select(
        folder_table.c.id,
        folder_table.c.parent_id,
    ).join(
        anc_alias, folder_table.c.id == anc_alias.c.parent_id
    )
    ancestors_cte = anchor.union_all(recursive_term)

    # Get all ancestor folder IDs (including the document's direct folders)
    all_ancestor_ids_result = await db.execute(select(ancestors_cte.c.id))
    all_ancestor_ids = [row[0] for row in all_ancestor_ids_result.all()]

    # Check if ANY FolderACL entries exist on any ancestor folder
    from sqlalchemy import func as sa_func
    acl_count_result = await db.execute(
        select(sa_func.count()).select_from(FolderACL).where(
            FolderACL.folder_id.in_(all_ancestor_ids),
            FolderACL.is_deleted == False,
        )
    )
    total_folder_acl_entries = acl_count_result.scalar()

    if total_folder_acl_entries == 0:
        return True  # No folder ACL entries anywhere in chain -> open access (backward compat)

    # Folder ACL entries exist — check user access
    # Check direct user entries
    user_acl_result = await db.execute(
        select(FolderACL).where(
            FolderACL.folder_id.in_(all_ancestor_ids),
            FolderACL.principal_id == user_id,
            FolderACL.principal_type == "user",
            FolderACL.is_deleted == False,
        )
    )
    for entry in user_acl_result.scalars().all():
        if has_sufficient_permission(PermissionLevel(entry.permission_level), required_level):
            return True

    # Check group entries
    from app.models.user import user_groups as ug_table
    group_result = await db.execute(select(ug_table.c.group_id).where(ug_table.c.user_id == user_id))
    group_ids = [row[0] for row in group_result.fetchall()]

    if group_ids:
        group_acl_result = await db.execute(
            select(FolderACL).where(
                FolderACL.folder_id.in_(all_ancestor_ids),
                FolderACL.principal_type == "group",
                FolderACL.principal_id.in_(group_ids),
                FolderACL.is_deleted == False,
            )
        )
        for entry in group_acl_result.scalars().all():
            if has_sufficient_permission(PermissionLevel(entry.permission_level), required_level):
                return True

    return False  # Folder ACL entries exist but user has no access

**1. Extend `get_folder_documents()` in `folder_service.py`:**

Add parameters `user_id: uuid.UUID | None = None` and `is_superuser: bool = False` to the function signature.

When `user_id` is provided and `is_superuser` is False, filter documents by ACL:
- Join to `DocumentACL` and `FolderACL` to determine visibility
- A document is visible if:
  a. It has direct ACL entries that grant access to the user (user or group principal), OR
  b. It has no direct ACL entries AND the folder chain has no FolderACL entries (open access), OR
  c. It has no direct ACL entries AND the folder chain has FolderACL entries that grant access to the user

For simplicity and to avoid duplicating the complex CTE logic, use a Python-level filter approach:
- Query all documents in the folder (existing logic)
- For each document, call `acl_service.check_permission()` to determine if user can see it
- This is acceptable because `get_folder_documents` is paginated (max 20 docs) and the CTE is efficient

Actually, for better performance, apply a subquery-based filter directly in SQL. The approach:

if user_id is not None and not is_superuser:
    from app.models.acl import DocumentACL, FolderACL
    from app.models.user import user_groups
    from sqlalchemy import or_, and_, exists, literal_column

    group_result = await db.execute(
        select(user_groups.c.group_id).where(user_groups.c.user_id == uid)
    )
    user_group_ids = [row[0] for row in group_result.all()]

    # Subquery: documents with direct ACL for this user
    has_direct_user_acl = exists(
        select(DocumentACL.id).where(
            DocumentACL.document_id == Document.id,
            DocumentACL.principal_id == uid,
            DocumentACL.principal_type == "user",
            DocumentACL.is_deleted == False,
        )
    )

    # Subquery: documents with direct ACL for user's groups
    has_direct_group_acl = exists(
        select(DocumentACL.id).where(
            DocumentACL.document_id == Document.id,
            DocumentACL.principal_type == "group",
            DocumentACL.principal_id.in_(user_group_ids),
            DocumentACL.is_deleted == False,
        )
    ) if user_group_ids else literal_column("0 = 1")

    # Subquery: documents with ANY direct ACL entries
    has_any_direct_acl = exists(
        select(DocumentACL.id).where(
            DocumentACL.document_id == Document.id,
            DocumentACL.is_deleted == False,
        )
    )

    # For documents without direct ACL, we check folder ACL via check_permission
    # But in SQL we approximate: doc has no direct ACL = use folder ACL logic
    # For the folder documents endpoint, we know the folder_id, so we can check
    # FolderACL on the ancestor chain of THIS folder.

    # Build ancestor CTE for the current folder
    folder_table = Folder.__table__
    anchor = (
        select(folder_table.c.id, folder_table.c.parent_id)

    folder_anc_cte = anchor.union_all(recursive)

    # Check if any FolderACL entries exist on any ancestor
    has_any_folder_acl = exists(
        select(FolderACL.id).where(
            FolderACL.folder_id.in_(select(folder_anc_cte.c.id)),
            FolderACL.is_deleted == False,
        )
    )

    # Check if user has folder ACL access
    has_user_folder_acl = exists(
